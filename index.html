// Q1. Bank Account Withdrawal let balance = 10000; let withdrawalAmount = 7000; let withdraw = new Promise((resolve, reject) => { if (balance >= withdrawalAmount) { resolve("Withdrawal Successful! Remaining Balance: Rs. " + (balance - withdrawalAmount)); } else { reject("Insufficient Balance!"); } }); withdraw .then((result) => console.log(result)) .catch((error) => console.log(error)); // Q2. Online Exam Result let marks = 75; let result = new Promise((resolve, reject) => { if (marks < 0 || marks > 100) { reject("Invalid Marks!"); } else if (marks >= 50) { resolve("Congratulations! You Passed."); } else { reject("Sorry! You Failed."); } }); result .then((message) => console.log(message)) .catch((error) => console.log(error)); // Q3. Product Delivery System let delivery = new Promise((resolve, reject) => { setTimeout(() => { let success = true; if (success) { resolve("Product Delivered Successfully"); } else { reject("Delivery Failed"); } }, 3000); }); delivery .then((message) => console.log(message)) .catch((error) => console.log(error)); // Q4. Promise Chaining let number = new Promise((resolve, reject) => { resolve(10); }); number .then((result) => { return result + 20; }) .then((result) => { return result * 2; }) .then((result) => { return result - 5; }) .then((finalResult) => { console.log("Final Result: " + finalResult); }) .catch((error) => console.log(error));
